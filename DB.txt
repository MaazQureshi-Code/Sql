-- Create Customers Table
CREATE TABLE customers (
    customer_id VARCHAR2(4) NOT NULL CONSTRAINT pk_customer PRIMARY KEY,
    name VARCHAR2(100) NOT NULL,
    email VARCHAR2(100) UNIQUE NOT NULL
);

-- Create Customer Details Table
CREATE TABLE customer_details (
    detail_id VARCHAR2(4) NOT NULL CONSTRAINT pk_customer_details PRIMARY KEY,
    customer_id VARCHAR2(4) NOT NULL,
    address VARCHAR2(255) NOT NULL,
    phone_number VARCHAR2(20) NOT NULL,
    CONSTRAINT fk_customer FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

-- Create Products Table
CREATE TABLE products (
    product_id VARCHAR2(4) NOT NULL CONSTRAINT pk_product PRIMARY KEY,
    name VARCHAR2(100) NOT NULL CONSTRAINT unique_product_name UNIQUE,
    category VARCHAR2(50) NOT NULL,
    price DECIMAL(10, 2) NOT NULL
);


-- Create Orders Table
CREATE TABLE orders (
    order_id VARCHAR2(4) NOT NULL CONSTRAINT pk_order PRIMARY KEY,
    customer_id VARCHAR2(4) NOT NULL,
    order_date DATE NOT NULL,
    total_amount DECIMAL(10, 2) NOT NULL,
    order_status VARCHAR2(20) CHECK (order_status IN ('Pending', 'Shipped', 'Delivered')) NOT NULL,
    CONSTRAINT fk_customer_order FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

-- Create Order Items Table
CREATE TABLE order_items (
    order_item_id VARCHAR2(4) NOT NULL CONSTRAINT pk_order_item PRIMARY KEY,
    order_id VARCHAR2(4) NOT NULL,
    product_id VARCHAR2(4) NOT NULL,
    quantity INT NOT NULL,
    CONSTRAINT fk_order FOREIGN KEY (order_id) REFERENCES orders(order_id),
    CONSTRAINT fk_product FOREIGN KEY (product_id) REFERENCES products(product_id)
);

-- Create Payments Table
CREATE TABLE payments (
    payment_id VARCHAR2(4) NOT NULL CONSTRAINT pk_payment PRIMARY KEY,
    order_id VARCHAR2(4) NOT NULL,
    payment_method VARCHAR2(20) CHECK (payment_method IN ('CreditCard', 'PayPal')) NOT NULL,
    payment_status VARCHAR2(20) CHECK (payment_status IN ('Completed', 'Pending', 'Failed')) NOT NULL,
    payment_date DATE NOT NULL,
    CONSTRAINT fk_order_payment FOREIGN KEY (order_id) REFERENCES orders(order_id)
);
